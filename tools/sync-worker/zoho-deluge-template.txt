// push_avalon_case — FINAL v2, case-level rows (field names confirmed 2026-07-27)
// -------------------------------------------------------------------------
// Zoho CRM workflow function, module "Avalon Assist" (API name: Cases).
// Rule: on Create or Edit → Instant Action → this function, recordId → #id.
// Category must be Automation. Set syncSecret to the worker's SYNC_SECRET.
//
// Pushes ONE identifier-free row for the changed case; the worker upserts it
// by case_key, so status changes overwrite the same row. Member fields
// (name, DOB, email, phone, government ID, dose, comments) are never read.

void automation.push_avalon_case(int recordId)
{
	syncUrl = "https://myrxcard-sync.trevorfallbacher.workers.dev";
	syncSecret = "PASTE-YOUR-SYNC-SECRET-HERE";

	rec = zoho.crm.getRecordById("Cases", recordId);

	cl = rec.get("Client_Name");
	clientName = "";
	if(cl != null)
	{
		clientName = ifnull(cl.get("name"),"").toString();
	}
	medName = ifnull(rec.get("Medication_Name"),"").toString().trim();
	created = ifnull(rec.get("Created_Time"),"").toString();
	if(clientName == "" || medName == "" || created.length() < 10)
	{
		info "skipped: record missing client, medication, or created time";
		return;
	}

	// canonical channel names (Source is free text with many spellings)
	normSource = Map();
	normSource.put("medsdirect","MedsDirect");
	normSource.put("meds direct","MedsDirect");
	normSource.put("canada outreach","Canada Outreach");
	normSource.put("canada outreached","Canada Outreach");
	normSource.put("nash","NASH");
	normSource.put("rxfree4me","RxFree4me");
	normSource.put("global rx","GlobalRx");
	normSource.put("globalrx","GlobalRx");
	srcRaw = ifnull(rec.get("Source"),"").toString().trim();
	source = ifnull(normSource.get(srcRaw.toLowerCase()), if(srcRaw == "","Direct",srcRaw));

	tpa = "";
	tp = rec.get("TPA");
	if(tp != null)
	{
		tpa = ifnull(tp.get("name"),"").toString();
	}
	closedDate = ifnull(rec.get("Closed_Date"),"").toString();
	if(closedDate.length() >= 10)
	{
		closedDate = closedDate.subString(0,10);
	}

	row = Map();
	row.put("case_key",recordId.toString());
	row.put("assist_number",ifnull(rec.get("Case_Number"),"").toString());
	row.put("client_name",clientName);
	row.put("tpa",tpa);
	row.put("group_number",ifnull(rec.get("Group_Number"),"").toString());
	row.put("source",source);
	row.put("status",ifnull(rec.get("Status"),"").toString());
	row.put("closed_reason",ifnull(rec.get("Closed_Reason"),"").toString());
	row.put("medication_name",medName);
	row.put("ndc",ifnull(rec.get("NDC"),"").toString().replaceAll("[^0-9]",""));
	row.put("medication_type",ifnull(rec.get("Medication_Type"),"").toString());
	row.put("month",created.subString(0,7));
	row.put("created_date",created.subString(0,10));
	row.put("closed_date",closedDate);
	mem = rec.get("Member");
	if(mem != null)
	{
		row.put("member_ref",ifnull(mem.get("id"),"").toString());
	}
	// age in whole years from Member_DOB — the DOB itself never leaves Zoho;
	// capped at 90 so the row stays HIPAA safe-harbor de-identified
	dobStr = ifnull(rec.get("Member_DOB"),"").toString();
	if(dobStr.length() >= 10)
	{
		dob = dobStr.subString(0,10).toDate("yyyy-MM-dd");
		todayD = zoho.currentdate;
		memberAge = todayD.getYear() - dob.getYear();
		if(todayD.getMonth() < dob.getMonth() || (todayD.getMonth() == dob.getMonth() && todayD.getDay() < dob.getDay()))
		{
			memberAge = memberAge - 1;
		}
		if(memberAge > 90)
		{
			memberAge = 90;
		}
		if(memberAge >= 0 && memberAge <= 90)
		{
			row.put("member_age",memberAge);
		}
	}
	if(rec.get("AWP") != null)
	{
		row.put("awp",rec.get("AWP").toDecimal());
	}
	if(rec.get("Avalon_Fee") != null)
	{
		row.put("avalon_fee",rec.get("Avalon_Fee").toDecimal());
	}
	if(rec.get("AA_Price") != null)
	{
		row.put("aa_price",rec.get("AA_Price").toDecimal());
	}
	if(rec.get("AA_Savings") != null)
	{
		row.put("aa_savings",rec.get("AA_Savings").toDecimal());
	}
	if(rec.get("Avalon_Savings") != null)
	{
		row.put("avalon_savings",rec.get("Avalon_Savings").toDecimal());
	}
	if(rec.get("MyRXCard_Pricing") != null)
	{
		row.put("myrxcard_pricing",rec.get("MyRXCard_Pricing").toDecimal());
	}
	if(rec.get("MEDSDirect_Pricing") != null)
	{
		row.put("medsdirect_pricing",rec.get("MEDSDirect_Pricing").toDecimal());
	}
	if(rec.get("RxFree4me_Pricing") != null)
	{
		row.put("rxfree4me_pricing",rec.get("RxFree4me_Pricing").toDecimal());
	}
	if(rec.get("GlobalRx_Pricing") != null)
	{
		row.put("globalrx_pricing",rec.get("GlobalRx_Pricing").toDecimal());
	}
	if(rec.get("Canada_Pharmacy_Export_Pricing") != null)
	{
		row.put("canada_pricing",rec.get("Canada_Pharmacy_Export_Pricing").toDecimal());
	}
	if(rec.get("Shipping_Fees") != null)
	{
		row.put("shipping_fees",rec.get("Shipping_Fees").toDecimal());
	}
	if(rec.get("Physician_Fees") != null)
	{
		row.put("physician_fees",rec.get("Physician_Fees").toDecimal());
	}
	if(rec.get("Other_Fees") != null)
	{
		row.put("other_fees",rec.get("Other_Fees").toDecimal());
	}

	payload = Map();
	payload.put("secret",syncSecret);
	payload.put("rows",{row});
	resp = invokeurl
	[
		url :syncUrl
		type :POST
		parameters:payload.toString()
		headers:{"content-type":"application/json"}
	];
	info resp;
}
