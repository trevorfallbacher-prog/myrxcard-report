void automation.push_covered_lives(Int accountId)
{
	syncUrl = "https://myrxcard-sync.trevorfallbacher.workers.dev";
	syncSecret = "PASTE-YOUR-SYNC-SECRET-HERE";
	acct = zoho.crm.getRecordById("Accounts",accountId);
	nm = ifnull(acct.get("Account_Name"),"").toString().trim();
	lv = acct.get("Covered_Lives");
	if(nm == "" || lv == null)
	{
		info "skipped: missing account name or covered lives";
		return;
	}
	livesMap = Map();
	livesMap.put(nm,lv.toLong());
	payload = Map();
	payload.put("secret",syncSecret);
	payload.put("covered_lives",livesMap);
	resp = invokeurl
	[
		url :syncUrl
		type :POST
		parameters:payload.toString()
		headers:{"content-type":"application/json"}
	];
	info resp;
}
