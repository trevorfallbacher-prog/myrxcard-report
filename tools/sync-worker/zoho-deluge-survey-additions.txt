// QBR survey + response-time additions to push_avalon_case (2026-08-24)
// -------------------------------------------------------------------------
// Paste this block into the existing Deluge function push_avalon_case,
// AFTER the last `row.put("other_fees", ...)` line and BEFORE the
// `payload = Map();` line. It adds satisfaction-survey scores, the yes/no
// survey picklists, and the response-time milestone dates.
//
// NOT included (by design): the free-text survey comment fields
// (Other_Survey_Comments, Medication_or_Shipping_Issues_Comments,
// Prescription_Not_Accurate_Comments). Those can contain PHI and the worker
// firewall rejects free text — comments are curated separately, never synced.

// --- satisfaction scores ---
if(rec.get("How_Likely_Are_You_To_Recommend_1_10") != null)
{
	row.put("nps_recommend",rec.get("How_Likely_Are_You_To_Recommend_1_10").toDecimal());
}
if(rec.get("How_was_your_overall_experience_1_5") != null)
{
	row.put("experience_score",rec.get("How_was_your_overall_experience_1_5").toDecimal());
}
// survey counts as completed once either core score is answered
if(rec.get("How_was_your_overall_experience_1_5") != null || rec.get("How_Likely_Are_You_To_Recommend_1_10") != null)
{
	row.put("survey_completed","yes");
}

// --- survey picklists (short Yes/No-style answers) ---
row.put("would_use_again",ifnull(rec.get("Would_you_consider_using_Assist_again_in_future"),"").toString());
row.put("rx_accurate",ifnull(rec.get("Was_prescription_accurate"),"").toString());
row.put("advocate_clear",ifnull(rec.get("Did_your_advocate_provide_clear_answers"),"").toString());
row.put("member_shipping_issue",ifnull(rec.get("Any_Issues_with_medication_or_shipping"),"").toString());

// --- response-time milestone dates (date-only, first 10 chars) ---
reqD = ifnull(rec.get("Request_Date"),"").toString();
if(reqD.length() >= 10)
{
	row.put("request_date",reqD.subString(0,10));
}
rxD = ifnull(rec.get("Rx_Script_Date"),"").toString();
if(rxD.length() >= 10)
{
	row.put("rx_script_date",rxD.subString(0,10));
}
awpD = ifnull(rec.get("AWP_Pricing_Date"),"").toString();
if(awpD.length() >= 10)
{
	row.put("awp_pricing_date",awpD.subString(0,10));
}
fillD = ifnull(rec.get("Initial_Fill_Date"),"").toString();
if(fillD.length() >= 10)
{
	row.put("initial_fill_date",fillD.subString(0,10));
}
refD = ifnull(rec.get("Refill_Date"),"").toString();
if(refD.length() >= 10)
{
	row.put("refill_date",refD.subString(0,10));
}
